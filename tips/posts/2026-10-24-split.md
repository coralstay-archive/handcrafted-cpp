vector<string> split(const string& input, string delimiter) {
    vector<string> result;
    auto start = 0;
    auto end = input.find(delimiter);
    while (end != string::npos) {
        result.push_back(input.substr(start, end - start));
        start = end + delimiter.size();

    }
    result.push_back(input.substr(start));
    return result;
}

vector<string> split(const string& input, string delimiter) {
    vector<string> result;
    auto start = 0;
    auto end = input.find(delimiter);
    while (end != string::npos) {
        result.push_back(input.substr(start, end - start));
        start = end + delimiter.size();
        end = input.find(delimiter, start);
    }
    result.push_back(input.substr(start));
    return result;

TEST(StringSplitTest, GeneralCommaSeparated) {
    string input = "apple,banana,cherry";
    vector<string> expected = {"apple", "banana", "cherry"};
    EXPECT_EQ(split(input, ","), expected);

TEST(StringSplitTest, BestCase_NoDelimiter) {

    vector<string> expected = {"single_word_without_delimiter"};
    EXPECT_EQ(split(input, ","), expected);

TEST(StringSplitTest, EmptyInput) {
    string input = "";
    vector<string> expected = {""};

    string input = ",,,,";
    vector<string> expected = {"", "", "", "", ""};
    EXPECT_EQ(split(input, ","), expected);

    string input;
    const int count = 10000;
    for (int i = 0; i < count; ++i) {
        input += "token,";

    input += "token"; // 총 10,001개 토큰

    auto result = split(input, ",");
    EXPECT_EQ(result.size(), count + 1);
    EXPECT_EQ(result[0], "token");
    EXPECT_EQ(result.back(), "token");
}

## 5. ⏱️ 복잡도 심층 계산 및 원리 분석 (Complexity & Principles)
### 5.1 시간 복잡도 (Time Complexity): `O(N) (단일 문자 구분자 기준, N = 원본 문자열의 길이)`
계산 근거 및 동작 원리:
- 1. string::find(delimiter, start)는 시작 인덱스를 이전 일치 위치 뒤로 전진시키며 선형 순회하므로, 구분자가 단일 문자이거나 총 비교 횟수가 입력 크기에 비례하여 전체 탐색 비용은 O(N)입니다.
- 2. string::substr(start, len)은 발견된 각 토큰 문자열을 새로 힙에 할당하여 복사합니다. 잘라낸 모든 토큰들의 길이 총합은 원본 문자열 길이 N과 동일하므로, 전체 루프에 걸친 substr 복사 시간의 합산 역시 O(N)입니다.
- 3. 따라서 전체 알고리즘의 시간 복잡도는 O(N)입니다.

### 5.2 공간 복잡도 (Space Complexity): `O(N) (결과 std::vector<string> 및 문자열 조각 복사 비용)`
계산 근거 및 메모리 구조:
- 1. 반환되는 std::vector<string>에 쪼개진 모든 토큰이 담기며, 저장되는 문자열 바이트의 총합은 원본 문자열 N과 같으므로 O(N) 공간을 차지합니다.
- 2. std::vector는 원소가 늘어남에 따라 기하급수적 재할당(Geometric Reallocation)을 수행하지만, 점근적 보조 공간 복잡도는 여전히 O(N)입니다.
- 3. 함수 내부에서 사용하는 인덱스 변수(start, end)는 O(1) 크기의 스택 변수입니다.

## 6. ⚖️ 최선의 경우 vs 최악의 경우 분석 (Best & Worst Cases)
### 6.1 🟢 최선의 경우 (Best Case)
- 발생 시나리오: 구분자(delimiter)가 문자열에 전혀 존재하지 않거나, 단 1개의 토큰만 있는 경우 (예: 'hello_world', 구분자 ',')
- 측정 복잡도: `O(N) 시간, O(N) 공간 (벡터 재할당 0회)`
- 원리 및 이유: find() 호출 1회 만에 string::npos를 반환하여 while 루프를 전혀 타지 않고, 원본 문자열 1개만 vector에 담아 즉시 반환되므로 루프 오버헤드와 vector 재할당이 최소화됩니다.
- GoogleTest 검증 케이스: `StringSplitTest.BestCase_NoDelimiter`

### 6.2 🔴 최악의 경우 (Worst Case)
- 발생 시나리오: 구분자가 연속으로 나타나 수많은 빈 토큰이 발생하는 경우 (예: ',,,,')
- 측정 복잡도: `O(N) 시간 + vector 힙 재할당(Reallocation) 오버헤드 극대화`
- 원리 및 병목 원인: 문자열 길이 N에 대해 매 문자마다 일치가 발생하여 N+1개의 string 객체가 생성됩니다. vector::push_back이 매 바이트마다 호출되어 동적 버퍼 확장과 빈번한 힙 할당이 누적되어 실행 시간이 크게 증가합니다.
- GoogleTest 검증 케이스: `StringSplitTest.WorstCase_ConsecutiveDelimiters, StringSplitTest.WorstCase_LargeScaleStress`
